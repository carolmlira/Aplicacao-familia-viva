function menuShow(){
    let menuMobile = document.querySelector('menu-mobile')
    if (menuMobile.classList.contains('open')){
        menuMobile.classList.remove('open')
        document.querySelector('#menu-icon').src ="assets/icons/density_medium_24dp_E8EAED_FILL0_wght400_GRAD0_opsz24.png"
    }else {
        menuMobile.classList.add('open')
        document.querySelector('#menu.icon'). src = "assets/icons/close.png"
    }
    
}
